mqclient

자기 응답 큐(/w13_rep.PID)를 먼저 만든 뒤 서버의 요청 큐에 요청을 넣고, mq_timedreceive로 최대 3초까지 답을 기다리는 클라이언트입니다. quit을 보내면 서버가 끝납니다.

준비 중…

/home/student/class13-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있고, 소스를 고쳐 make나 gcc로 다시 빌드할 수 있습니다 (처음 컴파일할 때 컴파일러를 내려받느라 잠시 걸립니다). 편집기는 vi입니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./mqserver & SRV=$!; sleep 1

    서버를 백그라운드로 띄우고 PID를 SRV에 적어 둔다.

  2. ./mqclient apple & ./mqclient banana & ./mqclient cherry; sleep 1

    클라이언트 셋이 동시에 묻는다. 응답 큐가 PID마다 달라 각자 자기 답(APPLE, BANANA, CHERRY)을 받는다.

  3. kill -STOP $SRV; time ./mqclient late; kill -CONT $SRV; sleep 1

    서버를 멈춰 두면 클라이언트는 3초 뒤 '응답 없음: mq_timedreceive: Connection timed out'으로 포기한다. 서버를 다시 깨우면 뒤늦게 처리하려다 '응답 큐 없음'을 기록한다.

  4. ./mqclient quit; wait

    quit 요청에는 응답이 없다. 서버가 요청 큐를 지우고 끝난다.

  5. ls /dev/mqueue

    요청 큐와 응답 큐가 모두 지워져 남은 것이 없다.

mqclient는 mqserver가 떠 있어야 동작하므로 첫 단계에서 서버를 함께 띄웁니다. 셋째 단계는 타임아웃을 기다리느라 약 4초 걸립니다.

/* mqclient.c — 요청 하나를 보내고 자기 응답 큐에서 답을 기다린다 (최대 3초)
 *   ./mqclient TEXT      (./mqclient quit 은 서버를 끝낸다)
 */
#include <fcntl.h>
#include <mqueue.h>
#include <stdio.h>
#include <string.h>
#include <time.h>
#include <unistd.h>
#include "proto.h"
 
int main(int argc, char *argv[])
{
    if (argc != 2) { fprintf(stderr, "사용법: %s TEXT\n", argv[0]); return 2; }
 
    struct request req = { .pid = getpid() };
    snprintf(req.text, sizeof req.text, "%s", argv[1]);
    int quit = !strcmp(req.text, "quit");       /* quit에는 응답이 없다 */
 
    /* ① 응답 큐를 먼저 만든다 — 요청을 보낸 뒤에 만들면 늦을 수 있다 */
    char name[32];
    reply_name(name, sizeof name, req.pid);
    struct mq_attr attr = { .mq_maxmsg = 1,
                            .mq_msgsize = sizeof(struct reply) };
    mqd_t repq = mq_open(name, O_CREAT | O_EXCL | O_RDONLY, 0622, &attr);
    if (repq == (mqd_t)-1) { perror("mq_open(응답 큐)"); return 1; }
 
    /* ② 서버의 요청 큐를 이름으로 찾아 요청을 넣는다 */
    int rc = 1;
    mqd_t reqq = mq_open(REQ_QUEUE, O_WRONLY);
    if (reqq == (mqd_t)-1) {
        perror("서버가 없다: mq_open(" REQ_QUEUE ")");
        goto out;
    }
    if (mq_send(reqq, (char *)&req, sizeof req, 0) == -1) {
        perror("mq_send");
        goto out;
    }
    if (quit) { rc = 0; goto out; }
 
    /* ③ 내 응답 큐에서 기다린다. 서버가 죽었을 수 있으므로 3초까지만 */
    struct timespec until;
    clock_gettime(CLOCK_REALTIME, &until);
    until.tv_sec += 3;
    struct reply rep;
    if (mq_timedreceive(repq, (char *)&rep, sizeof rep, NULL, &until) == -1) {
        perror("응답 없음: mq_timedreceive");
        goto out;
    }
    printf("[클라이언트 %d] #%d %s → %s\n", (int)req.pid, rep.seq, req.text,
           rep.text);
    rc = 0;
 
out:
    if (reqq != (mqd_t)-1) mq_close(reqq);
    mq_close(repq);
    mq_unlink(name);                    /* 내 응답 큐는 내가 지운다 */
    return rc;
}
proto.h 27줄 
/* proto.h — mqserver와 mqclient가 함께 쓰는 약속(프로토콜) */
#ifndef PROTO_H
#define PROTO_H
 
#include <stdio.h>
#include <sys/types.h>
 
#define REQ_QUEUE "/w13_req"            /* 서버의 요청 큐: 모두가 아는 이름 */
#define TEXT_MAX  64
 
struct request {                        /* 클라이언트 → 서버 */
    pid_t pid;                          /* 응답을 돌려받을 곳 */
    char  text[TEXT_MAX];
};
 
struct reply {                          /* 서버 → 클라이언트 */
    int  seq;                           /* 서버가 처리한 순번 */
    char text[TEXT_MAX];
};
 
/* 응답 큐의 이름은 클라이언트마다 다르다: /w13_rep.PID */
static inline void reply_name(char *buf, size_t size, pid_t pid)
{
    snprintf(buf, size, "/w13_rep.%d", (int)pid);
}
 
#endif
