mqserver

요청 큐 /w13_req 하나로 여러 클라이언트의 요청을 받아 대문자로 바꾼 뒤, 클라이언트마다 다른 응답 큐(/w13_rep.PID)로 돌려주는 서버입니다. SIGINT/SIGTERM을 받으면 큐를 지우고 끝납니다.

/home/student/class13-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있고, 소스를 고쳐 make나 gcc로 다시 빌드할 수 있습니다 (처음 컴파일할 때 컴파일러를 내려받느라 잠시 걸립니다). 편집기는 vi입니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./mqclient hello; echo "exit=$?"

    서버를 띄우기 전: 요청 큐가 없어 '서버가 없다' 오류로 끝난다 (exit=1).

  2. ./mqserver & sleep 1; ls -l /dev/mqueue

    서버가 /w13_req 큐를 만들고 mq_receive에서 기다린다. /dev/mqueue에 w13_req가 보인다.

  3. ./mqclient hello

    서버 기록 '[서버] #1 PID n: hello → HELLO' 와 클라이언트가 받은 답이 출력된다.

  4. ./mqclient world; ls /dev/mqueue

    순번이 #2로 올라간다. 클라이언트의 응답 큐(w13_rep.PID)는 클라이언트가 지워서 남아 있지 않다.

  5. kill -INT $!; wait; ls /dev/mqueue

    Ctrl-C 대신 SIGINT를 보낸다. mq_receive가 EINTR로 깨어나고 서버는 큐를 mq_unlink한 뒤 끝난다 ('[서버] 끝: /w13_req 지움').

터미널이 하나라 서버를 백그라운드(&)로 띄웁니다. $!는 마지막 백그라운드 프로세스(서버)의 PID이므로 단계 사이에 다른 백그라운드 명령을 실행하지 마세요.

/* mqserver.c — 요청 큐 하나로 여러 클라이언트의 요청을 받아 대문자로 바꿔 돌려준다
 *   ./mqserver        (끝내기: ./mqclient quit 또는 Ctrl-C)
 */
#include <ctype.h>
#include <errno.h>
#include <fcntl.h>
#include <mqueue.h>
#include <signal.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include "proto.h"
 
static volatile sig_atomic_t stop;
static void on_signal(int sig) { (void)sig; stop = 1; }
 
int main(void)
{
    struct sigaction sa = { .sa_handler = on_signal };  /* SA_RESTART 없음 */
    sigaction(SIGINT, &sa, NULL);
    sigaction(SIGTERM, &sa, NULL);
 
    struct mq_attr attr = { .mq_maxmsg = 10,
                            .mq_msgsize = sizeof(struct request) };
    mq_unlink(REQ_QUEUE);               /* 지난 실행이 남긴 큐를 치운다 */
    mqd_t reqq = mq_open(REQ_QUEUE, O_CREAT | O_EXCL | O_RDONLY, 0622, &attr);
    if (reqq == (mqd_t)-1) { perror("mq_open"); return 1; }
    fprintf(stderr, "[서버 %d] %s 에서 기다림\n", (int)getpid(), REQ_QUEUE);
 
    for (int seq = 1; !stop; seq++) {
        struct request req;
        if (mq_receive(reqq, (char *)&req, sizeof req, NULL) == -1) {
            if (errno == EINTR) continue;       /* 시그널: stop을 다시 본다 */
            perror("mq_receive");
            break;
        }
        req.text[TEXT_MAX - 1] = '\0';          /* 받은 것은 믿지 않는다 */
        if (!strcmp(req.text, "quit")) break;
 
        struct reply rep = { .seq = seq };
        for (int i = 0; i < TEXT_MAX; i++)
            rep.text[i] = (char)toupper((unsigned char)req.text[i]);
 
        char name[32];
        reply_name(name, sizeof name, req.pid);
        mqd_t repq = mq_open(name, O_WRONLY | O_NONBLOCK);
        if (repq == (mqd_t)-1) {                /* 클라이언트가 이미 떠났다 */
            fprintf(stderr, "[서버] #%d PID %d: 응답 큐 없음 (%s)\n", seq,
                    (int)req.pid, strerror(errno));
            continue;
        }
        if (mq_send(repq, (char *)&rep, sizeof rep, 0) == -1)
            perror("mq_send");
        mq_close(repq);
        fprintf(stderr, "[서버] #%d PID %d: %s → %s\n", seq, (int)req.pid,
                req.text, rep.text);
    }
 
    mq_close(reqq);
    mq_unlink(REQ_QUEUE);               /* 이름을 지워야 큐가 사라진다 */
    fprintf(stderr, "[서버] 끝: %s 지움\n", REQ_QUEUE);
    return 0;
}

/* proto.h — mqserver와 mqclient가 함께 쓰는 약속(프로토콜) */
#ifndef PROTO_H
#define PROTO_H
 
#include <stdio.h>
#include <sys/types.h>
 
#define REQ_QUEUE "/w13_req"            /* 서버의 요청 큐: 모두가 아는 이름 */
#define TEXT_MAX  64
 
struct request {                        /* 클라이언트 → 서버 */
    pid_t pid;                          /* 응답을 돌려받을 곳 */
    char  text[TEXT_MAX];
};
 
struct reply {                          /* 서버 → 클라이언트 */
    int  seq;                           /* 서버가 처리한 순번 */
    char text[TEXT_MAX];
};
 
/* 응답 큐의 이름은 클라이언트마다 다르다: /w13_rep.PID */
static inline void reply_name(char *buf, size_t size, pid_t pid)
{
    snprintf(buf, size, "/w13_rep.%d", (int)pid);
}
 
#endif
