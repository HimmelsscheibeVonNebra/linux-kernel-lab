file_copy

4096바이트 버퍼로 read한 만큼 write_all로 새 파일에 기록해 복사한다. 목적지는 O_CREAT | O_EXCL로 열기 때문에 기존 파일을 덮어쓰지 않는다.

준비 중…

/home/student/class03-02에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. printf 'ABCDEFGHIJ\n' > in.txt; rm -f copy.txt; ./file_copy in.txt copy.txt; echo "exit=$?"

    첫 복사는 성공한다(상태 0).

  2. cat copy.txt; wc -c in.txt copy.txt

    내용과 크기(11바이트)가 원본과 같다.

  3. ./file_copy in.txt copy.txt; echo "exit=$?"

    목적지가 이미 있으면 File exists로 실패(상태 1)하고 기존 파일은 보존된다.

  4. ./file_copy missing.txt new.txt; echo "exit=$?"; ls new.txt

    입력이 없으면 목적지를 만들기 전에 실패하므로 new.txt는 생기지 않는다.

  5. dd if=/dev/zero of=big.bin bs=1000 count=10 2>/dev/null; rm -f big.copy; strace -e trace=openat,read,write,close ./file_copy big.bin big.copy

    10000바이트를 4096, 4096, 1808, 0 순서로 읽고 같은 길이만큼 쓰는 것을 확인한다. 앞부분은 라이브러리 로딩이다.

  6. wc -c big.bin big.copy

    두 파일 모두 10000바이트다.

#include <fcntl.h>
#include <stdio.h>
#include <unistd.h>

#include "student/io_helpers.h"

#include "io_helpers.h"

/* Caller owns both descriptors. Copy bytes, not metadata. */
static int copy_fd(int in, int out)
{
    unsigned char buf[4096];
    for (;;) {
        ssize_t n = read(in, buf, sizeof buf);
        if (n == 0) return 0;
        if (n == -1) {
            if (errno == EINTR) continue;
            return -1;
        }
        if (write_all(out, buf, (size_t)n) == -1) return -1;
    }
}
int main(int argc, char **argv)
{
    if (argc != 3) {
        fprintf(stderr, "usage: %s INPUT NEW_OUTPUT\n", argv[0]);
        return 2;
    }
    int in = open(argv[1], O_RDONLY);
    if (in == -1) { perror(argv[1]); return 1; }
    int out = open(argv[2], O_WRONLY | O_CREAT | O_EXCL, 0666);
    if (out == -1) {
        perror(argv[2]);
        if (close(in) == -1) perror("close input");
        return 1;
    }
    int status = 0;
    if (copy_fd(in, out) == -1) {
        perror("copy"); status = 1;
    }
    if (close(in) == -1) { perror("close input"); status = 1; }
    if (close(out) == -1) { perror("close output"); status = 1; }
    return status;
}

#ifndef IO_HELPERS_H
#define IO_HELPERS_H
#include <errno.h>
#include <stddef.h>
#include <unistd.h>
/* Blocking descriptors. Retry interruptions and advance after partial writes. */
static int write_all(int fd, const void *data, size_t count)
{
    const unsigned char *p = data;
    while (count > 0) {
        ssize_t n = write(fd, p, count);
        if (n == -1) {
            if (errno == EINTR) continue;
            return -1;
        }
        if (n == 0) { errno = EIO; return -1; }
        p += (size_t)n;
        count -= (size_t)n;
    }
    return 0;
}
