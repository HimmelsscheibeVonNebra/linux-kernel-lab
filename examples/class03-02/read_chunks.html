read_chunks

4바이트 버퍼로 read를 반복하며 반환값을 stderr에, 읽은 데이터를 stdout에 내보낸다. 짧은 read와 EOF(0)를 구별해 본다.

준비 중…

/home/student/class03-02에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. printf ABCDEFGHIJ > in.txt; ./read_chunks in.txt; echo

    read 반환값 4, 4, 2, 0과 데이터 조각이 번갈아 나온다. 2도 성공이고 0이 EOF다.

  2. ./read_chunks in.txt > out.txt; wc -c in.txt out.txt

    stdout만 파일로 보내면 진단 메시지는 화면에 남고 데이터 10바이트만 복사된다.

  3. : > empty.txt; ./read_chunks empty.txt; echo "exit=$?"

    빈 파일은 첫 read가 바로 0을 반환한다.

  4. ./read_chunks in.txt > /dev/full; echo "exit=$?"

    /dev/full에 쓰면 ENOSPC가 발생해 write 오류 메시지와 종료 상태 1이 나온다.

소스

read_chunks.c 29줄 
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <unistd.h>
#include "io_helpers.h"
int main(int argc, char **argv)
{
    if (argc != 2) {
        fprintf(stderr, "usage: %s INPUT\n", argv[0]); return 2;
    }
    int fd = open(argv[1], O_RDONLY);
    if (fd == -1) { perror(argv[1]); return 1; }
    unsigned char buf[4];
    int status = 0;
    for (;;) {
        ssize_t n = read(fd, buf, sizeof buf);
        if (n == -1) {
            if (errno == EINTR) continue;
            perror("read"); status = 1; break;
        }
        fprintf(stderr, "read returned %zd\n", n);
        if (n == 0) break;
        if (write_all(STDOUT_FILENO, buf, (size_t)n) == -1) {
            perror("write"); status = 1; break;
        }
    }
    if (close(fd) == -1) { perror("close"); status = 1; }
    return status;
}
io_helpers.h 22줄 
#ifndef IO_HELPERS_H
#define IO_HELPERS_H
#include <errno.h>
#include <stddef.h>
#include <unistd.h>
/* Blocking descriptors. Retry interruptions and advance after partial writes. */
static int write_all(int fd, const void *data, size_t count)
{
    const unsigned char *p = data;
    while (count > 0) {
        ssize_t n = write(fd, p, count);
        if (n == -1) {
            if (errno == EINTR) continue;
            return -1;
        }
        if (n == 0) { errno = EIO; return -1; }
        p += (size_t)n;
        count -= (size_t)n;
    }
    return 0;
}
#endif
