Linux Lab/4주차 1차시/dir_list

dir_list

opendir/readdir/closedir로 디렉토리의 이름 목록을 읽습니다. . 과 .. 도 출력하며, errno를 0으로 초기화해 끝과 오류를 구분합니다.

준비 중…

/home/student/class04-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. rm -rf dirlist_lab && mkdir -p dirlist_lab/sub && printf 'hello\n' > dirlist_lab/a.txt && touch dirlist_lab/.hidden

    파일, 숨김 파일, 하위 디렉토리를 준비합니다.

  2. ./dir_list dirlist_lab

    . 과 .. , .hidden까지 모든 이름이 한 줄씩 나옵니다(순서는 정렬되지 않음).

  3. ls -a dirlist_lab

    ls -a 결과와 같은 이름 집합인지 비교합니다.

  4. ./dir_list dirlist_lab/a.txt; echo "status=$?"

    일반 파일은 ENOTDIR(Not a directory)로 실패합니다.

  5. ./dir_list dirlist_lab/missing; echo "status=$?"

    없는 경로는 ENOENT로 실패합니다.

_FILE_OFFSET_BITS=64 없이 빌드되어 d_ino/d_off가 32비트를 넘는 파일시스템에서는 readdir가 EOVERFLOW를 낼 수 있습니다. tmpfs에서는 문제가 없습니다.

소스

dir_list.c 49줄 
#define _POSIX_C_SOURCE 200809L
#include <dirent.h>
#include <errno.h>
#include <stdio.h>
 
int main(int argc, char **argv)
{
    if (argc != 2) {
        fprintf(stderr, "usage: %s DIRECTORY\n", argv[0]);
        return 2;
    }
 
    DIR *dir = opendir(argv[1]);
    if (dir == NULL) {
        perror(argv[1]);
        return 1;
    }
 
    int status = 0;
    for (;;) {
        /* NULL이 끝인지 오류인지 구분하기 위해 호출 직전에 초기화한다. */
        errno = 0;
        struct dirent *entry = readdir(dir);
        if (entry == NULL) {
            if (errno != 0) {
                perror("readdir");
                status = 1;
            }
            break;
        }
 
        /* 이름만 출력한다. 이 기초 예제는 .과 ..도 보여 준다. */
        if (puts(entry->d_name) == EOF) {
            perror("stdout");
            status = 1;
            break;
        }
    }
 
    if (closedir(dir) == -1) {
        perror("closedir");
        status = 1;
    }
    if (fflush(stdout) == EOF) {
        perror("stdout");
        status = 1;
    }
    return status;
}
