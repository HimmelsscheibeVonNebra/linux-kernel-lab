file_read

16바이트 버퍼로 read를 반복해 0(EOF)이 나올 때까지 읽고, 실제 읽은 바이트 수만큼만 표준 출력에 쓴다.

준비 중…

/home/student/class03-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. printf '0123456789abcdefghijklmnopqrstuv' > long.txt; ./file_read long.txt; echo

    32바이트 파일 내용이 그대로 출력된다.

  2. strace -e trace=read ./file_read long.txt > /dev/null

    마지막 세 read의 반환값 16, 16, 0을 찾는다. 0이 EOF다. 앞쪽 read는 라이브러리 로딩이다.

  3. ./file_read long.txt > copy.txt; wc -c long.txt copy.txt

    복사본 크기가 원본과 같은 32바이트인지 확인한다.

  4. printf 'A\000B' > binary.dat; ./file_read binary.dat | hexdump -C

    NUL 바이트(00)도 길이 기반으로 그대로 전달된다: 41 00 42.

  5. : > empty.txt; ./file_read empty.txt; echo "exit=$?"

    빈 파일은 첫 read가 0을 반환하며 정상 종료(0)한다.

  6. ./file_read missing.txt; echo "exit=$?"

    없는 파일은 open이 실패해 오류 메시지와 종료 상태 1이 나온다.

소스

file_read.c 30줄 
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <unistd.h>
#include "io_helpers.h"
 
int main(int argc, char *argv[])
{
    if (argc != 2) {
        fprintf(stderr, "Usage: %s FILE\n", argv[0]);
        return 2;
    }
    int fd = open(argv[1], O_RDONLY);
    if (fd == -1) { perror(argv[1]); return 1; }
    unsigned char buf[16]; /* Small on purpose: observe repeated reads. */
    int status = 0;
    for (;;) {
        ssize_t n = read(fd, buf, sizeof buf);
        if (n == 0) break;
        if (n == -1) {
            if (errno == EINTR) continue;
            perror("read"); status = 1; break;
        }
        if (write_all(STDOUT_FILENO, buf, (size_t)n) == -1) {
            perror("write stdout"); status = 1; break;
        }
    }
    if (close(fd) == -1) { perror("close"); status = 1; }
    return status;
}
io_helpers.h 22줄 
#ifndef IO_HELPERS_H
#define IO_HELPERS_H
#include <errno.h>
#include <stddef.h>
#include <unistd.h>
/* Blocking descriptors. Retry interruptions and advance after partial writes. */
static int write_all(int fd, const void *data, size_t count)
{
    const unsigned char *p = data;
    while (count > 0) {
        ssize_t n = write(fd, p, count);
        if (n == -1) {
            if (errno == EINTR) continue;
            return -1;
        }
        if (n == 0) { errno = EIO; return -1; }
        p += (size_t)n;
        count -= (size_t)n;
    }
    return 0;
}
#endif
