Linux Lab/5주차 1차시/sysprobe

sysprobe

uname(2)의 필드와 sysconf(3) 값(온라인 CPU 수, 페이지 크기, CLK_TCK, OPEN_MAX, ARG_MAX)을 출력합니다. sysconf의 -1 반환을 errno로 구분하는 계약을 보여 줍니다.

준비 중…

/home/student/class05-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./sysprobe

    sysname, release, machine과 cpus_online, pagesize, clk_tck, open_max, arg_max 값을 확인합니다.

  2. uname -a

    sysprobe의 각 필드가 uname 명령의 어느 열과 대응하는지 맞춥니다.

  3. uname -r; uname -m

    release와 machine 필드만 따로 비교합니다.

  4. ulimit -Sn

    open_max 값은 셸의 FD soft 한계와 같습니다.

  5. strace -e trace=uname ./sysprobe

    uname 시스템 호출과 커널이 채워 준 구조체를 확인합니다.

이 VM은 32비트 단일 CPU이므로 machine=i686, cpus_online=1이 나옵니다. getconf 명령은 없으므로 ulimit으로 교차 확인합니다. strace 출력에는 동적 로더가 호출한 uname이 함께 보일 수 있습니다.

소스

sysprobe.c 52줄 
#include <errno.h>
#include <inttypes.h>
#include <stdio.h>
#include <sys/utsname.h>
#include <unistd.h>
 
/* sysconf는 실패해도 errno를 바꾸지 않을 수 있으므로
   호출 직전에 0으로 만들고 -1의 의미를 나누어 본다. */
static int probe(const char *label, int name)
{
    errno = 0;
    long value = sysconf(name);
    if (value == -1) {
        if (errno != 0) {
            perror(label);
            return 1;
        }
        printf("%s=unlimited\n", label);
        return 0;
    }
    printf("%s=%ld\n", label, value);
    return 0;
}
 
int main(int argc, char **argv)
{
    if (argc != 1) {
        fprintf(stderr, "usage: %s\n", argv[0]);
        return 2;
    }
    struct utsname u;
    if (uname(&u) == -1) {
        perror("uname");
        return 1;
    }
    printf("sysname=%s\n", u.sysname);
    printf("nodename=%s\n", u.nodename);
    printf("release=%s\n", u.release);
    printf("version=%s\n", u.version);
    printf("machine=%s\n", u.machine);
    int status = 0;
    status |= probe("cpus_online", _SC_NPROCESSORS_ONLN);
    status |= probe("pagesize", _SC_PAGESIZE);
    status |= probe("clk_tck", _SC_CLK_TCK);
    status |= probe("open_max", _SC_OPEN_MAX);
    status |= probe("arg_max", _SC_ARG_MAX);
    if (fflush(stdout) == EOF) {
        perror("stdout");
        return 1;
    }
    return status;
}
