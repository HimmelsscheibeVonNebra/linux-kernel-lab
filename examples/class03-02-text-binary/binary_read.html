binary_read

fopen("rb")과 fread로 int 3개를 읽어 십진수로 출력한다. 항목이 부족하거나 뒤에 데이터가 더 있으면 오류로 처리한다.

준비 중…

/home/student/class03-02-text-binary에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있고, 소스를 고쳐 make나 gcc로 다시 빌드할 수 있습니다 (처음 컴파일할 때 컴파일러를 내려받느라 잠시 걸립니다). 편집기는 vi입니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. printf '\012\000\000\000\024\000\000\000\036\000\000\000' > numbers.bin; ./binary_read numbers.bin

    직접 만든 12바이트를 int 3개로 해석해 10 20 30을 출력한다.

  2. printf '\012\000\000\000\024\000' > short.bin; ./binary_read short.bin; echo "exit=$?"

    6바이트뿐이라 완전한 항목은 1개다. fread 반환값으로 부족을 감지한다(상태 1).

  3. cat numbers.bin > extra.bin; printf 'X' >> extra.bin; ./binary_read extra.bin; echo "exit=$?"

    세 번째 int 뒤에 남는 바이트가 있으면 형식 오류로 거부한다.

  4. ./binary_read /dev/stdin < numbers.bin

    파일 이름 대신 표준 입력 경로를 줘도 같은 결과가 나온다.

소스

binary_read.c 34줄 
#include <stdio.h>
 
int main(int argc, char *argv[])
{
    if (argc != 2) {
        fprintf(stderr, "Usage: %s FILE\n", argv[0]);
        return 2;
    }
    FILE *fp = fopen(argv[1], "rb");
    if (!fp) { perror(argv[1]); return 1; }
    int values[3];
    int status = 1;
    size_t n = fread(values, sizeof values[0], 3, fp);
    if (n != 3) {
        if (ferror(fp)) perror("fread");
        else fprintf(stderr, "EOF: read %zu of 3 complete items\n", n);
        goto done;
    }
    /* This teaching format has exactly three ints; reject trailing bytes. */
    int extra = fgetc(fp);
    if (ferror(fp)) { perror("fgetc"); goto done; }
    if (extra != EOF) {
        fprintf(stderr, "Unexpected data after the third int\n");
        goto done;
    }
    if (printf("%d %d %d\n", values[0], values[1], values[2]) < 0) {
        perror("printf"); goto done;
    }
    status = 0;
done:
    if (fclose(fp) == EOF) { perror("fclose"); status = 1; }
    if (fflush(stdout) == EOF) { perror("stdout"); status = 1; }
    return status;
}
