file_seek

lseek로 바이트 위치를 옮겨 수정 전 내용을 읽어 보여 주고, 같은 위치로 다시 이동해 TEXT를 덮어쓴다.

준비 중…

/home/student/class03-02-text-binary에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있고, 소스를 고쳐 make나 gcc로 다시 빌드할 수 있습니다 (처음 컴파일할 때 컴파일러를 내려받느라 잠시 걸립니다). 편집기는 vi입니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. printf 'ABCDEFGHIJ' > text.txt; ./file_seek text.txt 3 XY; echo

    수정 전 오프셋 3부터의 내용 DEFGHIJ가 출력된다(안내문은 stderr).

  2. cat text.txt; echo

    최종 내용은 ABCXYFGHIJ다. 길이는 변하지 않는다.

  3. ./file_seek text.txt 0 ZZ; echo; cat text.txt; echo

    맨 앞 2바이트가 ZZ로 바뀐다.

  4. ./file_seek text.txt 3x Q; echo "exit=$?"

    숫자가 아닌 오프셋은 종료 상태 2로 거부되고 파일은 그대로다.

  5. strace -e trace=openat,read,write,lseek,_llseek,close ./file_seek text.txt 3 XY

    위치 이동 → read → 위치 이동 → write 호출 순서를 확인한다.

32비트 환경이라 off_t가 32비트이므로 2147483647보다 큰 오프셋은 'Offset is too large for off_t'로 거부된다. i686 glibc에서는 lseek()가 strace에 _llseek로 보일 수 있다.

#include <errno.h>
#include <fcntl.h>
#include <inttypes.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include "io_helpers.h"
 
int main(int argc, char *argv[])
{
    if (argc != 4) {
        fprintf(stderr, "Usage: %s FILE BYTE_OFFSET TEXT\n", argv[0]);
        return 2;
    }
    char *end;
    errno = 0;
    intmax_t value = strtoimax(argv[2], &end, 10);
    if (errno != 0 || end == argv[2] || *end != '\0' || value < 0) {
        fprintf(stderr, "Offset must be a nonnegative integer\n");
        return 2;
    }
    off_t offset = (off_t)value;
    if ((intmax_t)offset != value) {
        fprintf(stderr, "Offset is too large for off_t\n");
        return 2;
    }
    /* Existing, seekable file. No O_TRUNC and no O_APPEND. */
    int fd = open(argv[1], O_RDWR);
    if (fd == -1) { perror(argv[1]); return 1; }
    int status = 1;
    if (lseek(fd, offset, SEEK_SET) == (off_t)-1) {
        perror("lseek before read"); goto done;
    }
    unsigned char buf[16];
    ssize_t n;
    do { n = read(fd, buf, sizeof buf); } while (n == -1 && errno == EINTR);
    if (n == -1) { perror("read"); goto done; }
    fprintf(stderr, "Before edit: %zd byte(s) from offset %jd\n", n, value);
    if (write_all(STDOUT_FILENO, buf, (size_t)n) == -1) {
        perror("write stdout"); goto done;
    }
    /* read advanced the offset: seek back before replacing bytes. */
    if (lseek(fd, offset, SEEK_SET) == (off_t)-1) {
        perror("lseek before write"); goto done;
    }
    if (write_all(fd, argv[3], strlen(argv[3])) == -1) {
        perror("write file"); goto done;
    }
    status = 0;
done:
    if (close(fd) == -1) { perror("close"); status = 1; }
    return status;
}
io_helpers.h 22줄 
#ifndef IO_HELPERS_H
#define IO_HELPERS_H
#include <errno.h>
#include <stddef.h>
#include <unistd.h>
/* Blocking descriptors. Retry interruptions and advance after partial writes. */
static int write_all(int fd, const void *data, size_t count)
{
    const unsigned char *p = data;
    while (count > 0) {
        ssize_t n = write(fd, p, count);
        if (n == -1) {
            if (errno == EINTR) continue;
            return -1;
        }
        if (n == 0) { errno = EIO; return -1; }
        p += (size_t)n;
        count -= (size_t)n;
    }
    return 0;
}
#endif
