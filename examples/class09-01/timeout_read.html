timeout_read

alarm으로 read에 시간 제한을 걸고, 핸들러 없음(기본 동작 종료)·핸들러 있음(EINTR)·SA_RESTART(read 재시작) 세 경우를 비교한다.

준비 중…

/home/student/class09-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. (sleep 3; echo hello) | ./timeout_read default 1; echo "exit=$?"

    핸들러가 없으면 SIGALRM 기본 동작으로 종료: exit=142 (128+14)

  2. (sleep 3; echo hello) | ./timeout_read norestart 1; echo "exit=$?"

    핸들러 실행 뒤 read가 -1 EINTR로 돌아온다(약 1초, exit=1)

  3. (sleep 3; echo hello) | ./timeout_read restart 1; echo "exit=$?"

    SA_RESTART면 read가 다시 기다려 3초 뒤 6바이트를 읽는다(alarms=1)

  4. ./timeout_read bogus 2; echo "exit=$?"

    잘못된 모드는 usage, 종료 상태 2

파이프 왼쪽의 sleep 3이 끝나야 줄이 끝나므로 각 단계는 약 3초 걸린다. 키보드 입력으로 해 보려면 ./timeout_read restart 5 를 실행하고 5초 뒤에 입력한다.

소스

timeout_read.c 69줄 
#include <errno.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <unistd.h>
 
/* alarm으로 read에 시간 제한을 건다.
   default  : 핸들러가 없다. SIGALRM의 기본 동작(종료)이 일어난다.
   norestart: SIGALRM을 잡는다. 핸들러가 끝나면 read가 -1, EINTR로 돌아온다.
   restart  : SA_RESTART를 켠다. 핸들러가 끝나면 read가 다시 기다린다. */
 
static volatile sig_atomic_t alarms = 0;
 
static void on_alarm(int sig)
{
    static const char msg[] = "  [handler] SIGALRM\n";
    (void)sig;
    ssize_t r = write(STDOUT_FILENO, msg, sizeof msg - 1);
    (void)r;                    /* 핸들러 안에서는 printf 대신 write */
    alarms++;
}
 
static double now(void)
{
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return ts.tv_sec + ts.tv_nsec / 1e9;
}
 
int main(int argc, char **argv)
{
    long sec = argc == 3 ? strtol(argv[2], NULL, 10) : 0;
    if (argc != 3 || sec < 1 || sec > 60 || (strcmp(argv[1], "default") != 0
        && strcmp(argv[1], "norestart") != 0 && strcmp(argv[1], "restart") != 0)) {
        fprintf(stderr, "usage: %s default|norestart|restart seconds(1-60)\n",
                argv[0]);
        return 2;
    }
    if (strcmp(argv[1], "default") != 0) {
        struct sigaction sa;
        memset(&sa, 0, sizeof sa);
        sa.sa_handler = on_alarm;
        sigemptyset(&sa.sa_mask);
        sa.sa_flags = strcmp(argv[1], "restart") == 0 ? SA_RESTART : 0;
        if (sigaction(SIGALRM, &sa, NULL) == -1) {
            perror("sigaction");
            return 1;
        }
    }
    printf("mode=%s alarm=%lds: waiting for input\n", argv[1], sec);
    fflush(stdout);
 
    char buf[256];
    double t0 = now();
    alarm((unsigned)sec);
    ssize_t n = read(STDIN_FILENO, buf, sizeof buf);
    int saved = errno;
    alarm(0);                   /* 남은 알람 취소 */
    double t = now() - t0;
    if (n == -1 && saved == EINTR)
        printf("read: -1 EINTR after %.2fs -> timeout\n", t);
    else if (n == -1)
        printf("read: -1 %s\n", strerror(saved));
    else
        printf("read: %zd bytes after %.2fs (alarms=%d)\n", n, t, (int)alarms);
    return n == -1;
}
