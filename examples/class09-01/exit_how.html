exit_how

자식을 exit·abort·SIGSEGV·SIGTERM 등으로 끝내고 부모가 waitpid의 status를 WIFEXITED/WIFSIGNALED/WIFSTOPPED/WIFCONTINUED로 해석해 raw 값과 함께 출력한다.

준비 중…

/home/student/class09-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./exit_how exit; ./exit_how term

    정상 종료는 status=3, raw 0x0300 / SIGTERM은 signal 15, raw 0x000f

  2. ./exit_how abort; ./exit_how segv

    SIGABRT(6)·SIGSEGV(11)로 종료된 자식의 raw status

  3. (ulimit -c unlimited; ./exit_how abort); ls -l core* 2>/dev/null; rm -f core*

    코어 한계를 풀면 ', core dumped'와 raw 0x0086(0x80 비트)이 보이는지 확인

  4. ./exit_how wait > wait.log & sleep 1; cat wait.log

    wait 모드: 자식이 외부 시그널을 기다린다(자식 PID 출력)

  5. c=$(sed -n 's/^child=\([0-9]*\) .*/\1/p' wait.log); kill -STOP $c; sleep 1; kill -CONT $c; sleep 1; kill -KILL $c; wait

    자식에게 STOP → CONT → KILL을 차례로 보낸다

  6. cat wait.log; rm -f wait.log

    stopped by signal 19 → continued → killed by signal 9 (raw 0x0009) 순서로 보고

기본 코어 한계(ulimit -c)가 0이면 abort·segv에 'core dumped'가 붙지 않고 raw가 0x0006·0x000b로 나온다. 코어 파일 생성 여부는 게스트 커널 설정(core_pattern, CONFIG_COREDUMP)에 따라 다르다.

#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/wait.h>
#include <unistd.h>
 
/* 자식이 어떻게 끝났는지 부모가 종료 상태(status)로 읽는다.
   exit : 종료 코드 3으로 정상 종료      abort: abort()로 SIGABRT
   segv : 널 포인터에 써서 SIGSEGV        term : raise(SIGTERM)
   wait : 외부에서 kill로 보낸 시그널을 기다린다(STOP·CONT도 보고).
   fork·waitpid의 정식 계약은 10주차에서 다룬다. 여기서는 관찰 도구다. */
 
static void child_action(const char *mode)
{
    if (strcmp(mode, "exit") == 0)
        _exit(3);
    if (strcmp(mode, "abort") == 0)
        abort();
    if (strcmp(mode, "segv") == 0) {
        volatile int *p = NULL;
        *p = 1;                 /* 커널이 SIGSEGV를 보낸다 */
    }
    if (strcmp(mode, "term") == 0)
        raise(SIGTERM);
    if (strcmp(mode, "wait") == 0) {
        printf("child=%ld waiting: kill -SIGNAL %ld\n",
               (long)getpid(), (long)getpid());
        fflush(stdout);
        for (;;)
            pause();
    }
    _exit(0);
}
 
static int report(pid_t child, int status)
{
    if (WIFEXITED(status)) {
        printf("child=%ld exited, status=%d\n",
               (long)child, WEXITSTATUS(status));
    } else if (WIFSIGNALED(status)) {
        int sig = WTERMSIG(status);
        printf("child=%ld killed by signal %d (%s)%s\n", (long)child,
               sig, strsignal(sig), WCOREDUMP(status) ? ", core dumped" : "");
    } else if (WIFSTOPPED(status)) {
        int sig = WSTOPSIG(status);
        printf("child=%ld stopped by signal %d (%s)\n",
               (long)child, sig, strsignal(sig));
    } else if (WIFCONTINUED(status)) {
        printf("child=%ld continued\n", (long)child);
    }
    printf("  raw status=0x%04x\n", (unsigned)status);
    fflush(stdout);
    return WIFEXITED(status) || WIFSIGNALED(status);
}
 
int main(int argc, char **argv)
{
    const char *modes[] = {"exit", "abort", "segv", "term", "wait"};
    int ok = 0;
    for (size_t i = 0; argc == 2 && i < sizeof modes / sizeof modes[0]; i++)
        if (strcmp(argv[1], modes[i]) == 0)
            ok = 1;
    if (!ok) {
        fprintf(stderr, "usage: %s exit|abort|segv|term|wait\n", argv[0]);
        return 2;
    }
    fflush(stdout);
    pid_t child = fork();
    if (child == -1) {
        perror("fork");
        return 1;
    }
    if (child == 0)
        child_action(argv[1]);
    for (;;) {
        int status;
        if (waitpid(child, &status, WUNTRACED | WCONTINUED) == -1) {
            perror("waitpid");
            return 1;
        }
        if (report(child, status))
            return 0;
    }
}
