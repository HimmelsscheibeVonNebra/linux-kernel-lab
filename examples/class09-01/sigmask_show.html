sigmask_show

/proc/PID/status의 SigPnd·ShdPnd·SigBlk·SigIgn·SigCgt 비트마스크를 시그널 이름으로 풀어 보여 준다(인자가 없으면 자기 자신).

준비 중…

/home/student/class09-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./sigmask_show

    자기 자신: 아무 핸들러도 등록하지 않아 대부분 비어 있다

  2. ./sigmask_show $$

    셸(ash)이 블록·무시·처리하는 시그널

  3. sleep 5 & ./sigmask_show $! | tail -2; kill $!; wait

    백그라운드 sleep의 SigIgn·SigCgt

  4. (trap '' INT; exec ./sigmask_show) | tail -2

    무시(SIG_IGN) 설정은 exec를 넘어 상속된다: SigIgn에 INT

  5. ./sigmask_show 1 2; echo "exit=$?"

    인자가 많으면 usage, 종료 상태 2

i386의 시그널 번호는 x86-64와 같아 이름 표가 그대로 맞는다. 셸이 bash가 아니라 BusyBox ash이므로 $$의 마스크는 교안과 다르다.

소스

sigmask_show.c 59줄 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
 
/* /proc/PID/status의 Sig* 줄을 읽어 비트를 시그널 이름으로 푼다.
   비트 n-1(오른쪽 끝이 비트 0)이 시그널 n이다. 인자가 없으면 self.
   이름 표는 x86-64 리눅스 번호 기준이고, 32 이상은 RT(실시간)로 적는다. */
 
static const char *names[32] = {
    "?", "HUP", "INT", "QUIT", "ILL", "TRAP", "ABRT", "BUS", "FPE",
    "KILL", "USR1", "SEGV", "USR2", "PIPE", "ALRM", "TERM", "STKFLT",
    "CHLD", "CONT", "STOP", "TSTP", "TTIN", "TTOU", "URG", "XCPU",
    "XFSZ", "VTALRM", "PROF", "WINCH", "IO", "PWR", "SYS"};
 
static void decode(const char *line)
{
    char label[16];
    unsigned long long mask;
    if (sscanf(line, "%15[^:]: %llx", label, &mask) != 2)
        return;
    printf("%-6s %016llx ", label, mask);
    if (mask == 0)
        printf(" -");
    for (int sig = 1; sig <= 64; sig++) {
        if (!(mask & (1ULL << (sig - 1))))
            continue;
        if (sig < 32)
            printf(" %s", names[sig]);
        else
            printf(" RT%d", sig);
    }
    printf("\n");
}
 
int main(int argc, char **argv)
{
    if (argc > 2) {
        fprintf(stderr, "usage: %s [pid]\n", argv[0]);
        return 2;
    }
    char path[64];
    snprintf(path, sizeof path, "/proc/%s/status", argc == 2 ? argv[1] : "self");
    FILE *fp = fopen(path, "r");
    if (fp == NULL) {
        perror(path);
        return 1;
    }
    char line[256];
    while (fgets(line, sizeof line, fp) != NULL) {
        if (strncmp(line, "Name:", 5) == 0 || strncmp(line, "Pid:", 4) == 0)
            fputs(line, stdout);
        else if (strncmp(line, "SigPnd:", 7) == 0 || strncmp(line, "ShdPnd:", 7) == 0
                 || strncmp(line, "SigBlk:", 7) == 0 || strncmp(line, "SigIgn:", 7) == 0
                 || strncmp(line, "SigCgt:", 7) == 0)
            decode(line);
    }
    fclose(fp);
    return 0;
}
