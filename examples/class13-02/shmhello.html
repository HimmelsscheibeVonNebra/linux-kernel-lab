shmhello

shm_open · ftruncate · mmap(MAP_SHARED)으로 이름 있는 공유 메모리 /w13_hello 에 글을 써 두고, 서로 무관한 다른 프로세스가 같은 이름으로 붙여 읽습니다.

준비 중…

/home/student/class13-02에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있고, 소스를 고쳐 make나 gcc로 다시 빌드할 수 있습니다 (처음 컴파일할 때 컴파일러를 내려받느라 잠시 걸립니다). 편집기는 vi입니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./shmhello write "hello shm"

    공유 메모리를 만들고 글을 써 둔 뒤 프로세스가 끝난다.

  2. ./shmhello read

    다른 프로세스(PID가 다르다)가 같은 이름으로 붙여 "hello shm"을 읽는다.

  3. ls -l /dev/shm/w13_hello; od -c /dev/shm/w13_hello | head -2

    공유 메모리 객체는 /dev/shm 아래의 4096바이트 파일로 보이고, 내용도 파일로 읽을 수 있다.

  4. ./shmhello rm; ./shmhello read; echo "exit=$?"

    shm_unlink로 이름을 지우면 다시 열 수 없다 (No such file or directory, exit=1).

소스

shmhello.c 45줄 
/* shmhello.c — 이름 있는 공유 메모리에 글을 써 두고, 다른 프로세스가 읽는다
 *   ./shmhello write TEXT   만들고(없으면) 글을 써 둔다
 *   ./shmhello read         붙여서 읽는다
 *   ./shmhello rm           이름을 지운다
 */
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <sys/mman.h>
#include <unistd.h>
 
#define NAME "/w13_hello"
#define SIZE 4096
 
int main(int argc, char *argv[])
{
    const char *cmd = argc > 1 ? argv[1] : "";
 
    if (!strcmp(cmd, "write") && argc == 3) {
        int fd = shm_open(NAME, O_CREAT | O_RDWR, 0600);   /* ① 이름으로 연다 */
        if (fd == -1) { perror("shm_open"); return 1; }
        if (ftruncate(fd, SIZE) == -1) { perror("ftruncate"); return 1; }  /* ② */
        char *p = mmap(NULL, SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
        if (p == MAP_FAILED) { perror("mmap"); return 1; } /* ③ 붙인다 */
        close(fd);                      /* 붙인 뒤에는 fd가 필요 없다 */
        snprintf(p, SIZE, "%s", argv[2]);       /* 그냥 메모리에 쓴다 */
        printf("[%d] 씀: \"%s\" (주소 %p)\n", (int)getpid(), p, (void *)p);
        munmap(p, SIZE);
    } else if (!strcmp(cmd, "read") && argc == 2) {
        int fd = shm_open(NAME, O_RDONLY, 0);
        if (fd == -1) { perror("shm_open"); return 1; }
        char *p = mmap(NULL, SIZE, PROT_READ, MAP_SHARED, fd, 0);
        if (p == MAP_FAILED) { perror("mmap"); return 1; }
        close(fd);
        printf("[%d] 읽음: \"%s\" (주소 %p)\n", (int)getpid(), p, (void *)p);
        munmap(p, SIZE);
    } else if (!strcmp(cmd, "rm") && argc == 2) {
        if (shm_unlink(NAME) == -1) { perror("shm_unlink"); return 1; }
        printf("지움: %s\n", NAME);
    } else {
        fprintf(stderr, "사용법: %s write TEXT | read | rm\n", argv[0]);
        return 2;
    }
    return 0;
}
