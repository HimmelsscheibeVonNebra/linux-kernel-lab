shmrace

공유 메모리 /w13_counter 의 count를 여러 프로세스가 잠금 없이 올려, 갱신이 사라지는 경쟁 상태를 관찰합니다.

준비 중…

/home/student/class13-02에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./shmrace rm 2>/dev/null; ./shmrace init

    공유 메모리를 만들고 count = 0으로 시작한다 (세마포어도 함께 초기화한다).

  2. ./shmrace add 1000000; ./shmrace show

    프로세스 하나만 올리면 정확히 count = 1000000 이다.

  3. for i in 1 2 3 4; do ./shmrace add 5000000 & done; wait

    프로세스 4개가 잠금 없이 500만 번씩 동시에 올린다.

  4. ./shmrace show

    잃은 것이 없다면 21000000 이어야 한다. 그보다 작으면 그만큼 갱신이 사라진 것이다.

  5. ./shmrace rm

    공유 메모리의 이름을 지운다.

이 VM은 CPU가 1개라서 경쟁은 프로세스가 '읽고 · 더하고 · 쓰는' 중간에 타이머 인터럽트로 교체될 때만 생깁니다. 그래서 반복 횟수를 크게(500만) 잡았지만 그래도 21000000이 그대로 나올 수 있습니다. 셋째·넷째 단계를 몇 번 반복해 보세요 (합계는 그만큼씩 늘어납니다). 32비트라 struct shared는 20바이트입니다.

shmrace.c 35줄 
/* shmrace.c — 공유 메모리의 값을 여러 프로세스가 잠금 없이 올린다
 *   ./shmrace init     공유 메모리를 만들고 count = 0
 *   ./shmrace add N    count를 N번 올린다 (잠금 없음)
 *   ./shmrace show     지금 값을 본다
 *   ./shmrace rm       이름을 지운다
 */
#include <string.h>
#include "counter.h"
 
int main(int argc, char *argv[])
{
    const char *cmd = argc > 1 ? argv[1] : "";
 
    if (!strcmp(cmd, "init") && argc == 2) {
        struct shared *s = attach(1);
        if (sem_init(&s->lock, 1, 1) == -1) { perror("sem_init"); return 1; }
        s->count = 0;
        printf("만듦: %s (%zu바이트), count = 0\n", SHM_NAME, sizeof *s);
    } else if (!strcmp(cmd, "add") && argc == 3) {
        struct shared *s = attach(0);
        long n = atol(argv[2]);
        for (long i = 0; i < n; i++)
            s->count++;                 /* 읽고 · 더하고 · 쓴다: 세 단계 */
        printf("[%d] %ld번 올림 → count = %ld\n", (int)getpid(), n, s->count);
    } else if (!strcmp(cmd, "show") && argc == 2) {
        printf("count = %ld\n", attach(0)->count);
    } else if (!strcmp(cmd, "rm") && argc == 2) {
        if (shm_unlink(SHM_NAME) == -1) { perror("shm_unlink"); return 1; }
        printf("지움: %s\n", SHM_NAME);
    } else {
        fprintf(stderr, "사용법: %s init | add N | show | rm\n", argv[0]);
        return 2;
    }
    return 0;
}
counter.h 36줄 
/* counter.h — shmrace와 semcount가 함께 쓰는 공유 메모리의 모양 */
#ifndef COUNTER_H
#define COUNTER_H
 
#include <fcntl.h>
#include <semaphore.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/mman.h>
#include <unistd.h>
 
#define SHM_NAME "/w13_counter"
 
struct shared {                 /* 공유 메모리의 맨 앞에 이 구조체가 놓인다 */
    sem_t lock;                 /* 2부에서 쓴다: 이진 세마포어 (1 = 열림) */
    volatile long count;        /* 여러 프로세스가 함께 올리는 값 */
};
 
/* 이름으로 열어 내 주소 공간에 붙인다. create이면 만들고 크기를 정한다 */
static inline struct shared *attach(int create)
{
    int flags = create ? O_CREAT | O_EXCL | O_RDWR : O_RDWR;
    int fd = shm_open(SHM_NAME, flags, 0600);
    if (fd == -1) { perror("shm_open"); exit(1); }
    if (create && ftruncate(fd, sizeof(struct shared)) == -1) {
        perror("ftruncate");
        exit(1);
    }
    struct shared *s = mmap(NULL, sizeof *s, PROT_READ | PROT_WRITE,
                            MAP_SHARED, fd, 0);
    if (s == MAP_FAILED) { perror("mmap"); exit(1); }
    close(fd);                  /* 매핑은 fd를 닫아도 남는다 */
    return s;
}
 
#endif
