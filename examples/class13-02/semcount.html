semcount

shmrace와 같은 공유 카운터를 프로세스 간 세마포어(sem_wait · sem_post)로 잠그고 올립니다. hold · try · value로 잠금이 쥐어진 상태, sem_timedwait의 시간 제한, 잠근 채 죽은 프로세스의 문제를 확인합니다.

준비 중…

/home/student/class13-02에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있고, 소스를 고쳐 make나 gcc로 다시 빌드할 수 있습니다 (처음 컴파일할 때 컴파일러를 내려받느라 잠시 걸립니다). 편집기는 vi입니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./shmrace rm 2>/dev/null; ./shmrace init

    공유 메모리와 세마포어(처음 값 1)를 shmrace init으로 만든다. semcount는 이것이 있어야 동작한다.

  2. for i in 1 2 3 4; do ./semcount add 200000 & done; wait; ./shmrace show

    프로세스 4개가 세마포어로 잠그고 올리면 항상 정확히 count = 800000 이다.

  3. ./semcount value

    아무도 쥐고 있지 않으면 세마포어 값 = 1 (열림).

  4. ./semcount hold 2 & sleep 1; ./semcount value; ./semcount try; wait

    hold가 쥔 동안 값은 0(잠김)이다. try는 기다리다가 hold가 2초 뒤 풀어 주면 잠그고 한 번 올린다 (count = 800001).

  5. ./semcount hold 8 & sleep 1; ./semcount try; kill $!; sleep 1; ./semcount value

    이번에는 2초 안에 풀리지 않아 try가 Connection timed out으로 포기한다. 쥐고 있던 프로세스를 kill로 죽여도 세마포어는 0(잠김)으로 남는다.

  6. ./shmrace rm

    잠긴 채 남은 공유 메모리를 지운다 (다시 쓰려면 shmrace init부터).

CPU가 1개여도 세마포어를 쓰면 결과는 항상 정확합니다. 다섯째 단계처럼 잠근 프로세스가 죽으면 세마포어는 자동으로 풀리지 않습니다.

/* semcount.c — 같은 값을 세마포어로 잠그고 올린다 (shmrace init으로 만든 뒤)
 *   ./semcount add N     잠그고 올리고 푼다 × N
 *   ./semcount hold SEC  잠근 채로 SEC초 동안 쥐고 있는다
 *   ./semcount try       2초까지만 기다렸다가 한 번 올린다
 *   ./semcount value     세마포어의 지금 값
 */
#include <string.h>
#include <time.h>
#include "counter.h"
 
int main(int argc, char *argv[])
{
    const char *cmd = argc > 1 ? argv[1] : "";
    struct shared *s = attach(0);
 
    if (!strcmp(cmd, "add") && argc == 3) {
        long n = atol(argv[2]);
        for (long i = 0; i < n; i++) {
            sem_wait(&s->lock);         /* 1 → 0. 이미 0이면 여기서 잔다 */
            s->count++;                 /* 임계 구역: 한 번에 한 프로세스만 */
            sem_post(&s->lock);         /* 0 → 1. 기다리던 프로세스를 깨운다 */
        }
        printf("[%d] %ld번 올림 → count = %ld\n", (int)getpid(), n, s->count);
    } else if (!strcmp(cmd, "hold") && argc == 3) {
        sem_wait(&s->lock);
        printf("[%d] 잠금: %s초 동안 쥔다\n", (int)getpid(), argv[2]);
        fflush(stdout);
        sleep((unsigned)atoi(argv[2])); /* 이 사이에 죽으면 잠긴 채로 남는다 */
        sem_post(&s->lock);
        printf("[%d] 풂\n", (int)getpid());
    } else if (!strcmp(cmd, "try") && argc == 2) {
        struct timespec until;
        clock_gettime(CLOCK_REALTIME, &until);
        until.tv_sec += 2;              /* 마감 시각 = 지금 + 2초 */
        if (sem_timedwait(&s->lock, &until) == -1) {
            perror("잠글 수 없다: sem_timedwait");
            return 1;
        }
        s->count++;
        sem_post(&s->lock);
        printf("[%d] 잠그고 한 번 올림 → count = %ld\n", (int)getpid(), s->count);
    } else if (!strcmp(cmd, "value") && argc == 2) {
        int v;
        sem_getvalue(&s->lock, &v);
        printf("세마포어 값 = %d (%s)\n", v, v > 0 ? "열림" : "잠김");
    } else {
        fprintf(stderr, "사용법: %s add N | hold SEC | try | value\n", argv[0]);
        return 2;
    }
    return 0;
}

/* counter.h — shmrace와 semcount가 함께 쓰는 공유 메모리의 모양 */
#ifndef COUNTER_H
#define COUNTER_H
 
#include <fcntl.h>
#include <semaphore.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/mman.h>
#include <unistd.h>
 
#define SHM_NAME "/w13_counter"
 
struct shared {                 /* 공유 메모리의 맨 앞에 이 구조체가 놓인다 */
    sem_t lock;                 /* 2부에서 쓴다: 이진 세마포어 (1 = 열림) */
    volatile long count;        /* 여러 프로세스가 함께 올리는 값 */
};
 
/* 이름으로 열어 내 주소 공간에 붙인다. create이면 만들고 크기를 정한다 */
static inline struct shared *attach(int create)
{
    int flags = create ? O_CREAT | O_EXCL | O_RDWR : O_RDWR;
    int fd = shm_open(SHM_NAME, flags, 0600);
    if (fd == -1) { perror("shm_open"); exit(1); }
    if (create && ftruncate(fd, sizeof(struct shared)) == -1) {
        perror("ftruncate");
        exit(1);
    }
    struct shared *s = mmap(NULL, sizeof *s, PROT_READ | PROT_WRITE,
                            MAP_SHARED, fd, 0);
    if (s == MAP_FAILED) { perror("mmap"); exit(1); }
    close(fd);                  /* 매핑은 fd를 닫아도 남는다 */
    return s;
}
 
#endif
