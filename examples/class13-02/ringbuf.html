ringbuf

공유 메모리에 놓은 8칸 원형 버퍼와 세마포어 셋(slots · items · mutex)으로 프로세스 사이의 생산자–소비자를 만듭니다. 가득 차면 생산자가, 비면 소비자가 잠듭니다.

준비 중…

/home/student/class13-02에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있고, 소스를 고쳐 make나 gcc로 다시 빌드할 수 있습니다 (처음 컴파일할 때 컴파일러를 내려받느라 잠시 걸립니다). 편집기는 vi입니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./ringbuf rm 2>/dev/null; ./ringbuf init; ./ringbuf stat

    버퍼와 세마포어를 만든다. 처음에는 빈 칸 8 · 든 칸 0.

  2. ./ringbuf put 1 12 & sleep 1; ./ringbuf stat

    생산자가 1~8을 넣고 9번째에서 빈 칸이 없어 잠든다. stat은 빈 칸 0 · 든 칸 8.

  3. ./ringbuf get 12; wait

    소비자가 꺼내기 시작하면 생산자가 깨어나 나머지(9~12)를 넣는다. 1~12가 순서대로 나오고 합 = 78.

  4. ./ringbuf get 3 & sleep 1; ./ringbuf put 7 3; wait

    반대로 빈 버퍼에서는 소비자가 잠든다. 7, 8, 9가 들어오자 깨어나 합 = 24를 출력한다.

  5. ./ringbuf stat; ./ringbuf rm

    모두 꺼냈으므로 다시 빈 칸 8 · 든 칸 0. 마지막에 이름을 지운다.

32비트라 struct ring은 88바이트로 출력됩니다 (64비트에서는 더 큽니다).

/* ringbuf.c — 공유 메모리의 원형 버퍼로 만드는 생산자–소비자
 *   ./ringbuf init                 버퍼(8칸)와 세마포어 셋을 만든다
 *   ./ringbuf put START COUNT [MS] START부터 COUNT개를 넣는다 (MS: 사이 쉬는 시간)
 *   ./ringbuf get COUNT [MS]       COUNT개를 꺼내 합을 출력한다
 *   ./ringbuf stat                 빈 칸 · 든 칸의 수
 *   ./ringbuf rm                   이름을 지운다
 */
#include <string.h>
#include "ring.h"
 
static void put(struct ring *r, long v)
{
    sem_wait(&r->slots);            /* 빈 칸이 생길 때까지 기다린다 */
    sem_wait(&r->mutex);            /* 버퍼를 잠근다 */
    r->buf[r->tail] = v;
    r->tail = (r->tail + 1) % RING_SIZE;
    sem_post(&r->mutex);
    sem_post(&r->items);            /* 든 칸 +1: 소비자를 깨운다 */
}
 
static long get(struct ring *r)
{
    sem_wait(&r->items);            /* 든 칸이 생길 때까지 기다린다 */
    sem_wait(&r->mutex);
    long v = r->buf[r->head];
    r->head = (r->head + 1) % RING_SIZE;
    sem_post(&r->mutex);
    sem_post(&r->slots);            /* 빈 칸 +1: 생산자를 깨운다 */
    return v;
}
 
int main(int argc, char *argv[])
{
    const char *cmd = argc > 1 ? argv[1] : "";
    setvbuf(stdout, NULL, _IOLBF, 0);
 
    if (!strcmp(cmd, "init") && argc == 2) {
        struct ring *r = ring_attach(1);
        sem_init(&r->slots, 1, RING_SIZE);  /* 둘째 인자 1: 프로세스 사이 공유 */
        sem_init(&r->items, 1, 0);
        sem_init(&r->mutex, 1, 1);
        r->head = r->tail = 0;
        printf("만듦: %s (%d칸, %zu바이트)\n", RING_NAME, RING_SIZE, sizeof *r);
    } else if (!strcmp(cmd, "put") && (argc == 4 || argc == 5)) {
        struct ring *r = ring_attach(0);
        long start = atol(argv[2]), n = atol(argv[3]);
        for (long i = 0; i < n; i++) {
            put(r, start + i);
            printf("[%d] 넣음 %ld\n", (int)getpid(), start + i);
            if (argc == 5) usleep((useconds_t)atoi(argv[4]) * 1000);
        }
    } else if (!strcmp(cmd, "get") && (argc == 3 || argc == 4)) {
        struct ring *r = ring_attach(0);
        long n = atol(argv[2]), sum = 0;
        for (long i = 0; i < n; i++) {
            long v = get(r);
            sum += v;
            printf("[%d] 꺼냄 %ld\n", (int)getpid(), v);
            if (argc == 4) usleep((useconds_t)atoi(argv[3]) * 1000);
        }
        printf("[%d] %ld개의 합 = %ld\n", (int)getpid(), n, sum);
    } else if (!strcmp(cmd, "stat") && argc == 2) {
        struct ring *r = ring_attach(0);
        int slots, items;
        sem_getvalue(&r->slots, &slots);
        sem_getvalue(&r->items, &items);
        printf("빈 칸 %d · 든 칸 %d\n", slots, items);
    } else if (!strcmp(cmd, "rm") && argc == 2) {
        if (shm_unlink(RING_NAME) == -1) { perror("shm_unlink"); return 1; }
        printf("지움: %s\n", RING_NAME);
    } else {
        fprintf(stderr, "사용법: %s init | put START COUNT [MS] | get COUNT [MS]"
                        " | stat | rm\n", argv[0]);
        return 2;
    }
    return 0;
}

/* ring.h — 공유 메모리에 놓는 원형 버퍼와 세마포어 셋 */
#ifndef RING_H
#define RING_H
 
#include <fcntl.h>
#include <semaphore.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/mman.h>
#include <unistd.h>
 
#define RING_NAME "/w13_ring"
#define RING_SIZE 8
 
struct ring {
    sem_t slots;                /* 빈 칸의 수: 처음에 RING_SIZE */
    sem_t items;                /* 든 칸의 수: 처음에 0 */
    sem_t mutex;                /* head · tail · buf를 지키는 잠금: 처음에 1 */
    int   head;                 /* 다음에 꺼낼 자리 */
    int   tail;                 /* 다음에 넣을 자리 */
    long  buf[RING_SIZE];
};
 
static inline struct ring *ring_attach(int create)
{
    int flags = create ? O_CREAT | O_EXCL | O_RDWR : O_RDWR;
    int fd = shm_open(RING_NAME, flags, 0600);
    if (fd == -1) { perror("shm_open"); exit(1); }
    if (create && ftruncate(fd, sizeof(struct ring)) == -1) {
        perror("ftruncate");
        exit(1);
    }
    struct ring *r = mmap(NULL, sizeof *r, PROT_READ | PROT_WRITE,
                          MAP_SHARED, fd, 0);
    if (r == MAP_FAILED) { perror("mmap"); exit(1); }
    close(fd);
    return r;
}
 
#endif
