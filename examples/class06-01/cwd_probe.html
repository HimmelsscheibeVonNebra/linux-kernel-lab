Linux Lab/6주차 1차시/cwd_probe

cwd_probe

getcwd를 버퍼 전달 방식과 (NULL, 0) 할당 방식 두 가지로 호출하고, 인자가 있으면 chdir(2) 후 다시 조회합니다. 실패 경로(ENOENT·ENOTDIR)도 확인합니다.

준비 중…

/home/student/class06-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./cwd_probe

    cwd와 cwd_alloc 두 방식의 결과가 같습니다.

  2. ./cwd_probe /tmp

    after_chdir=/tmp로 바뀝니다.

  3. pwd

    부모 셸의 작업 디렉토리는 그대로입니다.

  4. ./cwd_probe /no/such/dir; echo "status=$?"

    없는 경로는 'No such file or directory'와 종료 상태 1입니다.

  5. ./cwd_probe /proc/version; echo "status=$?"

    파일을 가리키면 'Not a directory'(ENOTDIR)로 실패합니다.

  6. readlink /proc/$$/cwd

    커널이 기록한 셸의 작업 디렉토리를 /proc에서 확인합니다.

README의 /etc/hosts는 게스트에 없을 수 있어 항상 존재하는 일반 파일 /proc/version으로 ENOTDIR을 확인합니다.

소스

cwd_probe.c 58줄 
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
 
/* getcwd는 상대 경로를 절대 경로로 푸는 기준인 작업 디렉터리를
   알려 준다. 호출 형태는 두 가지다. 호출자가 버퍼를 주거나,
   (NULL, 0)으로 크기를 정해 달라고 요청하거나. chdir은 이
   프로세스의 작업 디렉터리만 바꾼다. 부모 셸은 그대로다. */
static int show_cwd(const char *label)
{
    char buf[4096];
    if (getcwd(buf, sizeof buf) == NULL) {
        perror(label);
        return 1;
    }
    printf("%s=%s\n", label, buf);
    return 0;
}
 
static int show_cwd_alloc(const char *label)
{
    char *p = getcwd(NULL, 0);
    if (p == NULL) {
        perror(label);
        return 1;
    }
    printf("%s=%s\n", label, p);
    free(p);
    return 0;
}
 
int main(int argc, char **argv)
{
    if (argc > 2) {
        fprintf(stderr, "usage: %s [directory]\n", argv[0]);
        return 2;
    }
    if (show_cwd("cwd") != 0) {
        return 1;
    }
    if (show_cwd_alloc("cwd_alloc") != 0) {
        return 1;
    }
    if (argc == 2) {
        if (chdir(argv[1]) == -1) {
            perror(argv[1]);
            return 1;
        }
        if (show_cwd("after_chdir") != 0) {
            return 1;
        }
    }
    if (fflush(stdout) == EOF) {
        perror("stdout");
        return 1;
    }
    return 0;
}
