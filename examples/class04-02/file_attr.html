file_attr

기본은 lstat, --follow를 주면 stat으로 종류, inode, 링크 수, 권한(rwx 문자열), 소유자, 크기·블록 수, 나노초 단위 atime/mtime/ctime을 출력합니다.

준비 중…

/home/student/class04-02에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있고, 소스를 고쳐 make나 gcc로 다시 빌드할 수 있습니다 (처음 컴파일할 때 컴파일러를 내려받느라 잠시 걸립니다). 편집기는 vi입니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. rm -rf attr_lab && mkdir attr_lab && printf 'ABC\n' > attr_lab/note.txt && ln attr_lab/note.txt attr_lab/hard.txt && ln -s note.txt attr_lab/soft.txt

    4바이트 파일과 하드 링크, 심볼릭 링크를 준비합니다.

  2. ./file_attr attr_lab/note.txt

    type=regular, links=2, size=4와 rwx 문자열, 세 가지 시각을 확인합니다.

  3. ./file_attr attr_lab/soft.txt

    lstat 결과입니다. type=symlink, size는 'note.txt'의 길이 8입니다.

  4. ./file_attr --follow attr_lab/soft.txt

    stat은 링크를 따라가므로 대상의 type=regular, size=4와 같은 inode가 나옵니다.

  5. ln -s missing attr_lab/broken.txt && ./file_attr attr_lab/broken.txt && ./file_attr --follow attr_lab/broken.txt; echo "status=$?"

    깨진 링크도 링크 자체의 속성은 읽히지만 --follow 조회는 ENOENT로 실패합니다(상태 1).

  6. chmod 000 attr_lab/note.txt; ./file_attr attr_lab/note.txt; cat attr_lab/note.txt; chmod 600 attr_lab/note.txt

    권한이 0000이어도 속성 조회는 됩니다. 내용 읽기(cat)만 Permission denied로 실패합니다.

root가 아닌 student(uid 1000)로 실행하므로 mode 000 파일의 cat이 실제로 거부됩니다. _FILE_OFFSET_BITS=64로 빌드되어 크기는 64비트이지만 time_t는 32비트입니다. tmpfs라 blocks 값은 디스크 파일시스템과 다를 수 있습니다.

#include <inttypes.h>
#include <stdio.h>
#include <string.h>
#include <sys/stat.h>
 
static const char *kind(mode_t m)
{
    if (S_ISREG(m)) return "regular";
    if (S_ISDIR(m)) return "directory";
    if (S_ISLNK(m)) return "symlink";
    if (S_ISCHR(m)) return "char-device";
    if (S_ISBLK(m)) return "block-device";
    if (S_ISFIFO(m)) return "fifo";
    if (S_ISSOCK(m)) return "socket";
    return "unknown";
}
static void permissions(mode_t mode, char out[10])
{
    const mode_t bits[9] = {
        S_IRUSR, S_IWUSR, S_IXUSR, S_IRGRP, S_IWGRP, S_IXGRP,
        S_IROTH, S_IWOTH, S_IXOTH
    };
    for (int i = 0; i < 9; ++i)
        out[i] = (mode & bits[i]) ? "rwx"[i % 3] : '-';
    if (mode & S_ISUID) out[2] = (mode & S_IXUSR) ? 's' : 'S';
    if (mode & S_ISGID) out[5] = (mode & S_IXGRP) ? 's' : 'S';
    if (mode & S_ISVTX) out[8] = (mode & S_IXOTH) ? 't' : 'T';
    out[9] = '\0';
}
int main(int argc, char **argv)
{
    int follow = argc == 3 && strcmp(argv[1], "--follow") == 0;
    if (!(argc == 2 || follow)) {
        fprintf(stderr, "usage: %s [--follow] PATH\n", argv[0]);
        return 2;
    }
    const char *path = argv[follow ? 2 : 1];
    struct stat st;
    if ((follow ? stat(path, &st) : lstat(path, &st)) == -1) {
        perror(path); return 1;
    }
    char mode_text[10];
    permissions(st.st_mode, mode_text);
    printf("type=%s\n", kind(st.st_mode));
    printf("device=%ju inode=%ju links=%ju\n", (uintmax_t)st.st_dev,
           (uintmax_t)st.st_ino, (uintmax_t)st.st_nlink);
    printf("mode=%04jo rwx=%s uid=%ju gid=%ju\n",
           (uintmax_t)(st.st_mode & 07777), mode_text,
           (uintmax_t)st.st_uid, (uintmax_t)st.st_gid);
    printf("size=%jd blocks=%jd blksize=%jd\n", (intmax_t)st.st_size,
           (intmax_t)st.st_blocks, (intmax_t)st.st_blksize);
#ifdef __APPLE__
    struct timespec at = st.st_atimespec, mt = st.st_mtimespec, ct = st.st_ctimespec;
#else
    struct timespec at = st.st_atim, mt = st.st_mtim, ct = st.st_ctim;
#endif
    printf("atime=%jd.%09ld\n", (intmax_t)at.tv_sec, at.tv_nsec);
    printf("mtime=%jd.%09ld\n", (intmax_t)mt.tv_sec, mt.tv_nsec);
    printf("ctime=%jd.%09ld\n", (intmax_t)ct.tv_sec, ct.tv_nsec);
    if (fflush(stdout) == EOF) { perror("stdout"); return 1; }
    return 0;
}
