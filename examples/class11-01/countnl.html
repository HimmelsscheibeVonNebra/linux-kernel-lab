Linux Lab/11주차 1차시/countnl

countnl

같은 파일의 개행 수를 read(64 KiB 버퍼)와 mmap으로 세고, 경과 시간과 minor/major page fault 수(getrusage)를 출력합니다. drop은 posix_fadvise(DONTNEED)로 그 파일의 페이지 캐시를 비워 달라고 요청합니다.

준비 중…

/home/student/class11-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있고, 소스를 고쳐 make나 gcc로 다시 빌드할 수 있습니다 (처음 컴파일할 때 컴파일러를 내려받느라 잠시 걸립니다). 편집기는 vi입니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. seq 1 1000000 > lines.txt; ls -l lines.txt

    100만 줄(약 6.9 MB)짜리 파일을 만든다.

  2. ./countnl read lines.txt

    read 방식: lines=1000000, minflt는 거의 0이다 (커널이 버퍼로 복사해 준다).

  3. ./countnl mmap lines.txt

    mmap 방식: 같은 줄 수가 나오지만 페이지를 처음 만질 때마다 fault가 나서 minflt가 read보다 훨씬 많다.

  4. ./countnl mmap lines.txt seq

    madvise(MADV_SEQUENTIAL) 힌트를 준 경우의 시간과 fault 수를 비교한다.

  5. ./countnl drop lines.txt; ./countnl read lines.txt

    drop 뒤 다시 읽는다. 이 VM에서는 majflt가 여전히 0이다 (notes 참고).

  6. rm -f lines.txt

    실습 파일을 지운다 (tmpfs라서 파일이 곧 메모리다).

루트 파일시스템이 tmpfs(RAM)라서 파일 데이터가 곧 페이지 캐시입니다. posix_fadvise(DONTNEED)는 성공(0)을 돌려주지만 실제로 비워지지 않으므로 cold/warm 차이와 major fault는 관찰되지 않습니다. 또 RAM이 256 MB이므로 README의 247 MB 파일(seq 1 30000000) 실험은 할 수 없어 크기를 줄였습니다.

/* countnl.c — 같은 파일의 개행 수를 read 방식과 mmap 방식으로 세고
 *             경과 시간과 page fault 수를 함께 보여 준다.
 * 사용법:
 *   ./countnl read FILE
 *   ./countnl mmap FILE [seq]    seq: madvise(MADV_SEQUENTIAL)
 *   ./countnl drop FILE          이 파일의 페이지 캐시를 비운다 (cold 상태)
 */
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <time.h>
#include <sys/mman.h>
#include <sys/stat.h>
#include <sys/resource.h>
 
#define BUFSZ (64 * 1024)
 
static void die(const char *msg) { perror(msg); exit(1); }
 
static long count_buf(const char *p, size_t n)
{
    long c = 0;
    const char *end = p + n;
    while ((p = memchr(p, '\n', (size_t)(end - p))) != NULL) {
        c++;
        p++;
    }
    return c;
}
 
static long count_read(int fd)
{
    static char buf[BUFSZ];
    long c = 0;
    ssize_t n;
    while ((n = read(fd, buf, sizeof buf)) > 0)
        c += count_buf(buf, (size_t)n);
    if (n < 0) die("read");
    return c;
}
 
static long count_mmap(int fd, size_t len, int advice)
{
    if (len == 0) return 0;
    char *p = mmap(NULL, len, PROT_READ, MAP_PRIVATE, fd, 0);
    if (p == MAP_FAILED) die("mmap");
    if (advice != MADV_NORMAL && madvise(p, len, advice) < 0) die("madvise");
    long c = count_buf(p, len);
    munmap(p, len);
    return c;
}
 
static double now_ms(void)
{
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return ts.tv_sec * 1e3 + ts.tv_nsec / 1e6;
}
 
int main(int argc, char *argv[])
{
    if (argc < 3) {
        fprintf(stderr, "usage: %s read|mmap|drop FILE [seq]\n", argv[0]);
        return 2;
    }
    const char *mode = argv[1];
    int fd = open(argv[2], O_RDONLY);
    if (fd < 0) die(argv[2]);
    struct stat st;
    if (fstat(fd, &st) < 0) die("fstat");
 
    if (strcmp(mode, "drop") == 0) {
#ifdef POSIX_FADV_DONTNEED
        int e = posix_fadvise(fd, 0, 0, POSIX_FADV_DONTNEED);
        if (e != 0) { fprintf(stderr, "posix_fadvise: %s\n", strerror(e)); return 1; }
        printf("dropped page cache of %s\n", argv[2]);
        return 0;
#else
        fprintf(stderr, "drop: posix_fadvise is not available on this OS\n");
        return 1;
#endif
    }
 
    int seq = (argc == 4 && strcmp(argv[3], "seq") == 0);
    int advice = seq ? MADV_SEQUENTIAL : MADV_NORMAL;
    const char *tag = seq ? "+seq" : "";
    struct rusage r0, r1;
    getrusage(RUSAGE_SELF, &r0);
    double t0 = now_ms();
 
    long lines;
    if (strcmp(mode, "read") == 0)
        lines = count_read(fd);
    else if (strcmp(mode, "mmap") == 0)
        lines = count_mmap(fd, (size_t)st.st_size, advice);
    else { fprintf(stderr, "unknown mode %s\n", mode); return 2; }
 
    double t1 = now_ms();
    getrusage(RUSAGE_SELF, &r1);
    printf("%-4s%-4s lines=%ld  %8.1f ms  minflt=%ld  majflt=%ld\n",
           mode, tag, lines, t1 - t0,
           r1.ru_minflt - r0.ru_minflt, r1.ru_majflt - r0.ru_majflt);
    close(fd);
    return 0;
}
