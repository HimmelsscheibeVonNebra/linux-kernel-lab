hold_open

파일을 open한 채 지정한 초 동안 유지해 /proc/PID/fd에서 FD 표를 관찰하게 한다. open은 비어 있는 가장 작은 번호를 돌려준다.

준비 중…

/home/student/class07-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있고, 소스를 고쳐 make나 gcc로 다시 빌드할 수 있습니다 (처음 컴파일할 때 컴파일러를 내려받느라 잠시 걸립니다). 편집기는 vi입니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. echo hello > hold.txt

    열어 둘 파일 준비

  2. ./hold_open hold.txt 3 & sleep 1; ls -l /proc/$!/fd; wait

    0·1·2는 터미널, 3은 hold.txt를 가리킴. 끝나면 closed fd=3

  3. ./hold_open hold.txt 3 > out.txt & sleep 1; ls -l /proc/$!/fd; wait; cat out.txt

    리다이렉션하면 fd 1이 out.txt를 가리킨다

  4. ./hold_open hold.txt 1 <&-

    표준 입력을 닫고 실행하면 open이 0을 돌려준다(fd=0)

  5. ./hold_open /no/such/file 1; echo "exit=$?"

    open 실패: perror 메시지와 종료 상태 1

  6. rm -f hold.txt out.txt

소스

hold_open.c 60줄 
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
 
/* 파일을 열고 지정한 초 동안 유지하며 FD 표를 관찰하게 한다.
   open은 남은 가장 작은 번호를 돌려주므로 표준 스트림(0·1·2)
   만 열려 있으면 3이 돌아온다. 관찰 후 close로 반환한다.
   실패 경로에서도 연 FD는 닫는 것이 3주차 이후의 반복 원칙이다. */
 
int main(int argc, char **argv)
{
    if (argc < 2 || argc > 3) {
        fprintf(stderr, "usage: %s FILE [seconds 1-600]\n", argv[0]);
        return 2;
    }
    long seconds = 10;
    if (argc == 3) {
        char *end = NULL;
        seconds = strtol(argv[2], &end, 10);
        if (end == argv[2] || *end != '\0' ||
            seconds < 1 || seconds > 600) {
            fprintf(stderr, "usage: %s FILE [seconds 1-600]\n", argv[0]);
            return 2;
        }
    }
    int fd = open(argv[1], O_RDONLY);
    if (fd == -1) {
        perror(argv[1]);
        return 1;
    }
    printf("pid=%ld fd=%d path=%s\n", (long)getpid(), fd, argv[1]);
    printf("hold_seconds=%ld\n", seconds);
    printf("observe: ls -l /proc/%ld/fd\n", (long)getpid());
    printf("observe: ls /proc/%ld/fd | wc -l\n", (long)getpid());
    if (fflush(stdout) == EOF) {
        perror("stdout");
        close(fd);
        return 1;
    }
    for (long done = 1; done <= seconds; done++) {
        sleep(1);
        printf("hold=%ld of %ld\n", done, seconds);
        if (fflush(stdout) == EOF) {
            perror("stdout");
            close(fd);
            return 1;
        }
    }
    if (close(fd) == -1) {
        perror("close");
        return 1;
    }
    printf("closed fd=%d\n", fd);
    if (fflush(stdout) == EOF) {
        perror("stdout");
        return 1;
    }
    return 0;
}
