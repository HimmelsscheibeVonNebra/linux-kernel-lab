maps_show

/proc/self/maps를 getline으로 읽어 [heap]·[stack] 행을 출력하고 전체 구역 수와 .so 매핑 수를 센다.

준비 중…

/home/student/class07-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./maps_show

    heap_region·stack_region 행과 regions/so_regions 개수 확인

  2. cat /proc/self/maps | wc -l

    cat 자신의 구역 수와 비교(프로그램마다 다르다)

  3. ./maps_show x; echo "exit=$?"

    인자를 주면 usage, 종료 상태 2

구역 수는 32비트 glibc 환경 기준이라 교안(x86-64)의 숫자와 다를 수 있다.

소스

maps_show.c 66줄 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
 
/* /proc/self/maps는 이 프로세스의 가상 주소 공간을 행 단위로 보여
   준다. 각 행은 "시작-끝 권한 오프셋 장치 아이노드 경로" 형식이다.
   힙([heap])과 스택([stack]) 구역, 공유 라이브러리(.so) 매핑을
   골라 표시하고 전체 구역 수를 센다. /proc가 없는 환경에서는
   unsupported로 알린다. Linux에서 열기에 실패하면 실제 오류를
   보고한다. .so 매핑 수는 라이브러리 파일의 개수와 다르다. */
 
int main(int argc, char **argv)
{
    if (argc != 1) {
        fprintf(stderr, "usage: %s\n", argv[0]);
        return 2;
    }
    FILE *fp = fopen("/proc/self/maps", "r");
    if (fp == NULL) {
#ifdef __linux__
        perror("/proc/self/maps");
        return 1;
#else
        printf("maps=unsupported (no /proc on this system)\n");
        if (fflush(stdout) == EOF) {
            perror("stdout");
            return 1;
        }
        return 0;
#endif
    }
    char *line = NULL;
    size_t cap = 0;
    ssize_t len;
    long total = 0, heap = 0, stack = 0, so = 0;
    while ((len = getline(&line, &cap, fp)) != -1) {
        total++;
        if (strstr(line, "[heap]") != NULL) {
            heap++;
            printf("heap_region: %s", line);
        } else if (strstr(line, "[stack]") != NULL) {
            stack++;
            printf("stack_region: %s", line);
        } else if (strstr(line, ".so") != NULL) {
            so++;
        }
    }
    if (ferror(fp)) {
        perror("getline");
        free(line);
        fclose(fp);
        return 1;
    }
    printf("regions=%ld heap_regions=%ld stack_regions=%ld"
           " so_regions=%ld\n", total, heap, stack, so);
    free(line);
    if (fclose(fp) == EOF) {
        perror("fclose");
        return 1;
    }
    if (fflush(stdout) == EOF) {
        perror("stdout");
        return 1;
    }
    return 0;
}
