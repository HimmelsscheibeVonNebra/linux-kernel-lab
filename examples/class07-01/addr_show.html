Linux Lab/7주차 1차시/addr_show

addr_show

전역(data·bss)·정적·지역(stack)·malloc(heap) 변수의 가상 주소를 출력하고, 지정한 초 동안 살아 있어 /proc/PID/maps와 대조할 수 있게 한다.

준비 중…

/home/student/class07-01에서 student 사용자로 시작합니다. 컴파일된 실행 파일과 소스가 이미 들어 있습니다. 새로고침하거나 재부팅하면 처음 상태로 돌아갑니다.

실행 순서

  1. ./addr_show

    data·bss·stack·heap 주소가 서로 다른 구역에 있음을 확인

  2. ./addr_show | grep heap; ./addr_show | grep heap

    실행할 때마다 주소 값이 달라지는지(ASLR) 비교

  3. ./addr_show 3 & sleep 1; grep -E 'addr_show|heap|stack' /proc/$!/maps; wait

    출력된 주소가 maps의 어느 행(실행 파일·[heap]·[stack])에 속하는지 대조

  4. ./addr_show extra; echo "exit=$?"

    잘못된 인자는 usage와 종료 상태 2

32비트(i686) 게스트라 주소가 8자리 16진수이고 스택은 3GB 경계 아래(0xbf…/0xff… 부근)에 나타난다. 교안의 x86-64 주소 값과는 다르다.

소스

addr_show.c 59줄 
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
 
/* Linux의 일반적인 C 실행 환경에서 변수의 가상 주소를 관찰한다.
   [seconds] 동안 살아 있게 하여 같은 PID의 maps와 대조한다.
   함수 주소의 void * 변환이나 별개 객체의 포인터 대소 비교는
   하지 않는다. 코드 영역은 maps의 실행 권한(x)으로 찾는다. */
int global_init = 7;
int global_zero;
 
int main(int argc, char **argv)
{
    long seconds = 0;
    if (argc > 2) {
        fprintf(stderr, "usage: %s [seconds 0-600]\n", argv[0]);
        return 2;
    }
    if (argc == 2) {
        char *end = NULL;
        seconds = strtol(argv[1], &end, 10);
        if (end == argv[1] || *end != '\0' ||
            seconds < 0 || seconds > 600) {
            fprintf(stderr, "usage: %s [seconds 0-600]\n", argv[0]);
            return 2;
        }
    }
    int local = 3;
    static int local_static = 9;
    char *heap1 = malloc(64);
    char *heap2 = malloc(64);
    if (heap1 == NULL || heap2 == NULL) {
        perror("malloc");
        free(heap1);
        free(heap2);
        return 1;
    }
    printf("pid=%ld\n", (long)getpid());
    printf("data=%p bss=%p\n",
           (void *)&global_init, (void *)&global_zero);
    printf("data_static=%p stack=%p\n",
           (void *)&local_static, (void *)&local);
    printf("pointer_on_stack=%p\n", (void *)&heap1);
    printf("heap1=%p heap2=%p\n", (void *)heap1, (void *)heap2);
    printf("observe: cat /proc/%ld/maps\n", (long)getpid());
    printf("hold_seconds=%ld\n", seconds);
    if (fflush(stdout) == EOF) {
        perror("stdout");
        free(heap1);
        free(heap2);
        return 1;
    }
    unsigned int remaining = (unsigned int)seconds;
    while (remaining != 0)
        remaining = sleep(remaining);
    free(heap1);
    free(heap2);
    return 0;
}
