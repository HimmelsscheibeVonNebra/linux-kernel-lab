커널 모듈 실습

브라우저 안에서 부팅한 Linux 커널에 모듈을 로드·제거하고 로그, /proc, /sys, /dev를 확인합니다.

준비 중…

root 셸로 시작합니다. 모듈은 /root/*.ko, 소스는 /root/src에 있습니다. 올린 파일은 /mnt/host에 나타나며, 같은 커널로 빌드한 .ko를 올려 insmod 할 수 있습니다.

01 · 첫 커널 모듈

모듈을 로드·제거하고 커널 로그에서 init/exit 메시지를 확인합니다.

  1. uname -a

    32비트 x86 실습 커널인지 확인

  2. insmod hello.ko

    hello_init() 실행 — Hello from Linux Kernel!

  3. lsmod

    로드된 모듈 목록에 hello가 보입니다

  4. rmmod hello

    hello_exit() 실행 — Goodbye from Linux Kernel!

  5. dmesg | tail -n 5

    커널 로그 버퍼에서 두 메시지를 다시 확인

02 · procfs

/proc/kernellab 항목을 만들고 읽을 때마다 커널 함수가 호출되는 것을 봅니다.

  1. insmod proc_demo.ko

    /proc/kernellab 생성

  2. cat /proc/kernellab

    읽을 때마다 kernellab_show()가 실행됩니다

  3. cat /proc/kernellab

    read_count가 1 증가

  4. rmmod proc_demo

    항목 제거

03 · sysfs

kobject와 속성 파일로 커널 변수 하나를 사용자 공간에 노출합니다.

  1. insmod sysfs_demo.ko

    /sys/kernel/kernellab/value 생성

  2. cat /sys/kernel/kernellab/value

    value_show() — 초기값 0

  3. echo 42 > /sys/kernel/kernellab/value

    value_store()가 값을 바꿉니다

  4. cat /sys/kernel/kernellab/value

    42

  5. rmmod sysfs_demo

    kobject 제거

04 · 문자 디바이스

문자 디바이스를 등록하고 /dev/kernellab에 쓴 내용을 다시 읽습니다.

  1. insmod chardev_demo.ko

    장치 번호 할당, /dev/kernellab 자동 생성

  2. ls -l /dev/kernellab

    문자 디바이스(c)와 major/minor 번호

  3. echo hello > /dev/kernellab

    lab_write()가 버퍼에 저장

  4. cat /dev/kernellab

    lab_read()가 버퍼를 돌려줍니다

  5. rmmod chardev_demo

    장치 해제

#include <linux/init.h>
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/proc_fs.h>
#include <linux/seq_file.h>
#include <linux/jiffies.h>
#include <linux/atomic.h>
 
MODULE_LICENSE("GPL");
MODULE_AUTHOR("Kernel Lab");
MODULE_DESCRIPTION("procfs example: /proc/kernellab");
 
#define PROC_NAME "kernellab"
 
static atomic_t read_count = ATOMIC_INIT(0);
 
/* Called every time user space reads /proc/kernellab. */
static int kernellab_show(struct seq_file *m, void *v)
{
	seq_puts(m, "Hello from /proc/kernellab\n");
	seq_printf(m, "jiffies:    %lu\n", jiffies);
	seq_printf(m, "HZ:         %d\n", HZ);
	seq_printf(m, "read_count: %d\n", atomic_inc_return(&read_count));
	return 0;
}
 
static int __init proc_demo_init(void)
{
	if (!proc_create_single(PROC_NAME, 0444, NULL, kernellab_show))
		return -ENOMEM;
 
	pr_info("proc_demo: created /proc/%s\n", PROC_NAME);
	return 0;
}
 
static void __exit proc_demo_exit(void)
{
	remove_proc_entry(PROC_NAME, NULL);
	pr_info("proc_demo: removed /proc/%s\n", PROC_NAME);
}
 
module_init(proc_demo_init);
module_exit(proc_demo_exit);

#include <linux/init.h>
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/kobject.h>
#include <linux/sysfs.h>
 
MODULE_LICENSE("GPL");
MODULE_AUTHOR("Kernel Lab");
MODULE_DESCRIPTION("sysfs example: /sys/kernel/kernellab/value");
 
static struct kobject *lab_kobj;
static int value;
 
/* cat /sys/kernel/kernellab/value */
static ssize_t value_show(struct kobject *kobj, struct kobj_attribute *attr,
			  char *buf)
{
	return sysfs_emit(buf, "%d\n", value);
}
 
/* echo 42 > /sys/kernel/kernellab/value */
static ssize_t value_store(struct kobject *kobj, struct kobj_attribute *attr,
			   const char *buf, size_t count)
{
	int ret = kstrtoint(buf, 10, &value);
 
	if (ret)
		return ret;
 
	pr_info("sysfs_demo: value set to %d\n", value);
	return count;
}
 
static struct kobj_attribute value_attr = __ATTR_RW(value);
 
static int __init sysfs_demo_init(void)
{
	int ret;
 
	lab_kobj = kobject_create_and_add("kernellab", kernel_kobj);
	if (!lab_kobj)
		return -ENOMEM;
 
	ret = sysfs_create_file(lab_kobj, &value_attr.attr);
	if (ret) {
		kobject_put(lab_kobj);
		return ret;
	}
 
	pr_info("sysfs_demo: created /sys/kernel/kernellab/value\n");
	return 0;
}
 
static void __exit sysfs_demo_exit(void)
{
	kobject_put(lab_kobj);
	pr_info("sysfs_demo: removed /sys/kernel/kernellab\n");
}
 
module_init(sysfs_demo_init);
module_exit(sysfs_demo_exit);

#include <linux/init.h>
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/mutex.h>
#include <linux/uaccess.h>
 
MODULE_LICENSE("GPL");
MODULE_AUTHOR("Kernel Lab");
MODULE_DESCRIPTION("Character device example: /dev/kernellab");
 
#define DEV_NAME "kernellab"
#define BUF_SIZE 256
 
static dev_t devno;
static struct cdev lab_cdev;
static struct class *lab_class;
 
static char buffer[BUF_SIZE];
static size_t data_len;
static DEFINE_MUTEX(lab_lock);
 
static int lab_open(struct inode *inode, struct file *file)
{
	pr_info("chardev_demo: open\n");
	return 0;
}
 
static int lab_release(struct inode *inode, struct file *file)
{
	pr_info("chardev_demo: release\n");
	return 0;
}
 
/* cat /dev/kernellab : return what was written last */
static ssize_t lab_read(struct file *file, char __user *ubuf, size_t count,
			loff_t *ppos)
{
	ssize_t ret;
 
	mutex_lock(&lab_lock);
	ret = simple_read_from_buffer(ubuf, count, ppos, buffer, data_len);
	mutex_unlock(&lab_lock);
	return ret;
}
 
/* echo hi > /dev/kernellab : replace the stored message */
static ssize_t lab_write(struct file *file, const char __user *ubuf,
			 size_t count, loff_t *ppos)
{
	size_t len = min_t(size_t, count, BUF_SIZE);
 
	mutex_lock(&lab_lock);
	if (copy_from_user(buffer, ubuf, len)) {
		mutex_unlock(&lab_lock);
		return -EFAULT;
	}
	data_len = len;
	mutex_unlock(&lab_lock);
 
	pr_info("chardev_demo: stored %zu bytes\n", len);
	return count;
}
 
static const struct file_operations lab_fops = {
	.owner   = THIS_MODULE,
	.open    = lab_open,
	.release = lab_release,
	.read    = lab_read,
	.write   = lab_write,
};
 
static int __init chardev_demo_init(void)
{
	struct device *dev;
	int ret;
 
	ret = alloc_chrdev_region(&devno, 0, 1, DEV_NAME);
	if (ret)
		return ret;
 
	cdev_init(&lab_cdev, &lab_fops);
	ret = cdev_add(&lab_cdev, devno, 1);
	if (ret)
		goto err_region;
 
	lab_class = class_create(DEV_NAME);
	if (IS_ERR(lab_class)) {
		ret = PTR_ERR(lab_class);
		goto err_cdev;
	}
 
	/* devtmpfs creates /dev/kernellab automatically */
	dev = device_create(lab_class, NULL, devno, NULL, DEV_NAME);
	if (IS_ERR(dev)) {
		ret = PTR_ERR(dev);
		goto err_class;
	}
 
	pr_info("chardev_demo: registered /dev/%s (major %d, minor %d)\n",
		DEV_NAME, MAJOR(devno), MINOR(devno));
	return 0;
 
err_class:
	class_destroy(lab_class);
err_cdev:
	cdev_del(&lab_cdev);
err_region:
	unregister_chrdev_region(devno, 1);
	return ret;
}
 
static void __exit chardev_demo_exit(void)
{
	device_destroy(lab_class, devno);
	class_destroy(lab_class);
	cdev_del(&lab_cdev);
	unregister_chrdev_region(devno, 1);
	pr_info("chardev_demo: unregistered /dev/%s\n", DEV_NAME);
}
 
module_init(chardev_demo_init);
module_exit(chardev_demo_exit);
